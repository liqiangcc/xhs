SQL 复习实现：

~~~sql
WITH base AS (
    SELECT key_col, metric_col
    FROM source_table
    WHERE filter_col = :filter_value
), ranked AS (
    SELECT base.*,
           ROW_NUMBER() OVER (PARTITION BY key_col ORDER BY metric_col DESC) AS rn
    FROM base
)
SELECT *
FROM ranked
WHERE rn = 1;
~~~
